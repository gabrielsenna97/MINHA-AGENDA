filaSync.push({ acao:"upsert", tipo:tipo, registro: JSON.parse( JSON.stringify( registro ) ) }); salvarFila(); agendarLote(); } /* ============================================================ EXCLUSÃO ============================================================ */ function adicionarExclusaoFila( tipo, registroId ){ /* * Remove qualquer alteração * anterior desse registro. */ filaSync = filaSync.filter( item => !( item.tipo === tipo && String( item.registro?.id || item.id ) === String(registroId) ) ); filaSync.push({ acao:"delete", tipo:tipo, id:registroId }); salvarFila(); agendarLote(); } /* ============================================================ BLOCO DE NOTAS ============================================================ */ function adicionarBlocoFila(texto){ /* * Mantém somente o último texto. */ filaSync = filaSync.filter( item => item.acao !== "bloco" ); filaSync.push({ acao:"bloco", texto:texto }); salvarFila(); agendarLote(); } /* ============================================================ SALVAR FILA ============================================================ */ function salvarFila(){ localStorage.setItem( "minha_agenda_fila_sync", JSON.stringify( filaSync ) ); } /* ============================================================ AGENDAR ENVIO ============================================================ */ function agendarLote(){ clearTimeout( timerLote ); /* * Espera 800 ms. * * Assim, se o usuário cadastrar * várias coisas rapidamente, * tudo vai em uma única chamada. */ timerLote = setTimeout( enviarLote, 800 ); /* * Copia a fila atual. */ const lote = filaSync.slice(); try{ const resposta = await fetch( API_URL, { method:"POST", headers:{ "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ action: "batchSync", operacoes: lote }) } ); if( !resposta.ok ){ throw new Error( "HTTP "+ resposta.status ); } const resultado = await resposta.json(); if( !resultado.ok ){ throw new Error( resultado.erro || "Erro Google" ); } /* * Só remove da fila * aquilo que realmente * foi enviado. */ filaSync = filaSync.slice( lote.length ); salvarFila(); syncStatus( "☁️ Sincronizado" ); }catch(erro){ console.error( erro ); syncStatus( "⚠️ Aguardando internet", "erro" ); /* * Tenta novamente. */ setTimeout( enviarLote, 5000 ); }finally{ sincronizando = false; } } /* ============================================================ SUBSTITUIR enviarGoogle ============================================================ */ function enviarGoogle( tipo, registro ){ adicionarFila( tipo, registro ); } /* ============================================================ SUBSTITUIR excluirGoogle ============================================================ */ function excluirGoogle( tipo, registroId ){ adicionarExclusaoFila( tipo, registroId ); } /* ============================================================ SUBSTITUIR enviarBlocoGoogle ============================================================ */ function enviarBlocoGoogle( texto ){ adicionarBlocoFila( texto ); } /* ============================================================ BLOCO DE NOTAS ============================================================ */ function salvarBloco(){ dados.bloco = document.getElementById( "blocoTexto" ).value; salvar(); /* * Tela atualiza imediatamente. */ syncStatus( "⚡ Bloco salvo" ); /* * Google recebe em segundo plano. */ enviarBlocoGoogle( dados.bloco ); } /* ============================================================ INTERNET ============================================================ */ window.addEventListener( "online", function(){ syncStatus( "☁️ Conectado" ); enviarLote(); } ); window.addEventListener( "offline", function(){ syncStatus( "📱 Offline", "erro" ); } ); /* ============================================================ SINCRONIZAÇÃO AUTOMÁTICA ============================================================ */ setInterval( function(){ if( navigator.onLine && !sincronizando && filaSync.length ){ enviarLote(); } }, 10000 );
